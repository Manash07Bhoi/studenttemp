   246→    } else if (payload.type === 'inbox:expired') {

   492→// ---------- Inline Global Search Dialog ----------
   493→function GlobalSearchInline() {
   494→  const open = useAppStore((s) => s.globalSearchOpen)
   495→  const setOpen = useAppStore((s) => s.setGlobalSearchOpen)
   496→  const setActiveInboxId = useAppStore((s) => s.setActiveInboxId)
   497→  const setActiveSection = useAppStore((s) => s.setActiveSection)
   498→  const setOpenMessageId = useAppStore((s) => s.setOpenMessageId)
   499→  const [query, setQuery] = useState('')
   500→
   501→  const { data, isFetching } = useQuery({
   502→    queryKey: ['global-search', query],
   503→    queryFn: async () => {
   504→      if (!query || query.length < 2) return { results: [] as Array<Record<string, unknown>>, total: 0 }
   505→      const res = await fetch(`/api/search?q=${encodeURIComponent(query)}`)
   506→      if (!res.ok) throw new Error('Search failed')
   507→      return res.json() as Promise<{ results: Array<Record<string, unknown>>; total: number }>
   508→    },
   509→    enabled: query.length >= 2,
   510→    staleTime: 10_000,
   511→  })
   512→
   513→  const handleSelect = useCallback((result: Record<string, unknown>) => {
   514→    setActiveInboxId(result.inboxId as string)
   515→    setActiveSection('messages')
   516→    setTimeout(() => setOpenMessageId(result.id as string), 100)
   517→    setOpen(false)
   518→  }, [setActiveInboxId, setActiveSection, setOpenMessageId, setOpen])
   519→
   520→  const results = data?.results || []
   521→
   522→  return (
   523→    <AnimatePresence>
   524→      {open && (
   525→        <motion.div
   526→          initial={{ opacity: 0 }} animate={{ opacity: 1 }} exit={{ opacity: 0 }}
   527→          className="fixed inset-0 z-50 flex items-start justify-center pt-[10vh] px-4 bg-black/50 backdrop-blur-sm"
   528→          onClick={() => setOpen(false)}
   529→        >
   530→          <motion.div
   531→            initial={{ opacity: 0, scale: 0.96, y: -12 }}
   532→            animate={{ opacity: 1, scale: 1, y: 0 }}
   533→            exit={{ opacity: 0, scale: 0.96, y: -12 }}
   534→            transition={{ type: 'spring', stiffness: 400, damping: 32 }}
   535→            className="w-full max-w-2xl rounded-2xl border border-border/60 bg-card shadow-2xl overflow-hidden"
   536→            onClick={(e) => e.stopPropagation()}
   537→            role="dialog" aria-label="Global search"
   538→          >
   539→            <div className="flex items-center gap-3 border-b border-border/60 px-4 py-3">
   540→              <Search className="h-5 w-5 text-muted-foreground shrink-0" />
   541→              <input
   542→                type="text" value={query}
   543→                onChange={(e) => setQuery(e.target.value)}
   544→                placeholder="Search across all your inboxes…"
   545→                className="flex-1 bg-transparent text-base outline-none placeholder:text-muted-foreground"
   546→                autoFocus autoComplete="off" spellCheck={false}
   547→              />
   548→              {query && (
   549→                <button onClick={() => setQuery('')} className="text-muted-foreground hover:text-foreground" aria-label="Clear">
   550→                  <svg width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="currentColor" strokeWidth="2" strokeLinecap="round" strokeLinejoin="round"><path d="M18 6 6 18M6 6l12 12"/></svg>
   551→                </button>
   552→              )}
   553→            </div>
   554→            <div className="max-h-[60vh] overflow-y-auto scrollbar-thin">
   555→              {query.length < 2 ? (
   556→                <div className="px-4 py-12 text-center text-sm text-muted-foreground">
   557→                  Type at least 2 characters to search across all your inboxes.
   558→                </div>
   559→              ) : isFetching ? (
   560→                <div className="px-4 py-8 text-center text-sm text-muted-foreground">Searching…</div>
   561→              ) : results.length === 0 ? (
   562→                <div className="px-4 py-12 text-center text-sm text-muted-foreground">No messages found for "{query}"</div>
   563→              ) : (
   564→                <div className="py-2">
   565→                  <div className="px-4 py-2 text-xs font-semibold uppercase tracking-wider text-muted-foreground">
   566→                    {results.length} {results.length === 1 ? 'result' : 'results'}
   567→                  </div>
   568→                  {results.map((result) => (
   569→                    <button
   570→                      key={result.id as string}
   571→                      onClick={() => handleSelect(result)}
   572→                      className="w-full text-left px-4 py-3 hover:bg-accent/50 transition-colors"
   573→                    >
   574→                      <div className="flex items-start gap-3">
   575→                        <div className="grid h-9 w-9 shrink-0 place-items-center rounded-full bg-gradient-to-br from-emerald-400 to-cyan-500 text-white font-bold text-xs">
   576→                          {(result.fromName as string)?.[0]?.toUpperCase() || '?'}
   577→                        </div>
   578→                        <div className="flex-1 min-w-0">
   579→                          <div className="flex items-center justify-between gap-2">
   580→                            <span className="text-sm font-medium truncate">{highlightMatch(result.fromName as string, query)}</span>
   581→                            <span className="text-[11px] text-muted-foreground shrink-0 font-mono">{result.inboxEmail as string}</span>
   582→                          </div>
   583→                          <p className="mt-0.5 text-sm truncate font-semibold">{highlightMatch(result.subject as string, query)}</p>
   584→                          <p className="mt-0.5 text-xs text-muted-foreground truncate">{highlightMatch(result.previewText as string, query)}</p>
   585→                        </div>
   586→                      </div>
   587→                    </button>
   588→                  ))}
   589→                </div>
   590→              )}
   591→            </div>
   592→          </motion.div>
   593→        </motion.div>
   594→      )}
   595→    </AnimatePresence>
   596→  )
   597→}

   599→// ---------- Search result highlighting ----------
   600→function highlightMatch(text: string, query: string): ReactNode {
   601→  if (!query || !text) return text
   602→  const lower = text.toLowerCase()
   603→  const q = query.toLowerCase()
   604→  const idx = lower.indexOf(q)
   605→  if (idx === -1) return text
   606→  return (
   607→    <>
   608→      {text.slice(0, idx)}
   609→      <mark className="bg-emerald-500/25 text-emerald-800 dark:text-emerald-200 rounded px-0.5 font-semibold">
   610→        {text.slice(idx, idx + q.length)}
   611→      </mark>
   612→      {text.slice(idx + q.length)}
   613→    </>
   614→  )
   615→}
