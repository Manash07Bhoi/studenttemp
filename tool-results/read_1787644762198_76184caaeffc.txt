    45→    45→function sanitizeHtml(html: string): string {
    46→    46→  if (!html) return '<p style="color:#999;font-family:Arial">(empty body)</p>'
    47→    47→  let out = html
    48→    48→  out = out.replace(/<script[\s\S]*?<\/script>/gi, '')
    49→    49→  out = out.replace(/<style[\s\S]*?<\/style>/gi, '')
    50→    50→  out = out.replace(/\son\w+="[^"]*"/gi, '')
    51→    51→  out = out.replace(/\son\w+='[^']*'/gi, '')
    52→    52→  out = out.replace(/javascript:/gi, '')
    53→    53→  out = out.replace(/(<img[^>]+)src=["']https?:\/\/[^"']*["']/gi, '$1data-src-blocked="true"')
    54→    54→  out = out.replace(/<link[^>]+href=["']https?:\/\/[^"']*["'][^>]*>/gi, '')
    55→    55→  out = out.replace(/<iframe[\s\S]*?<\/iframe>/gi, '')
    56→    56→  return `<base target="_blank"><style>body{font-family:Arial,Helvetica,sans-serif;color:#1a1a1a;background:#fff;line-height:1.6;margin:0;padding:12px}a{color:#0ea5e9}img{max-width:100%;height:auto}</style>${out}`
    57→    57→}

    59→    59→function formatTime(iso: string): string {
    60→    60→  const d = new Date(iso)
    61→    61→  const now = new Date()
    62→    62→  const diff = now.getTime() - d.getTime()
    63→    63→  if (diff < 60_000) return 'just now'
    64→    64→  if (diff < 3_600_000) return `${Math.floor(diff / 60_000)}m ago`
    65→    65→  if (d.toDateString() === now.toDateString()) return d.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })
    66→    66→  return d.toLocaleDateString([], { month: 'short', day: 'numeric' })
    67→    67→}

   602→   602→// ---------------- Message List Item ----------------
   603→   603→// Swipe gestures per MOTION-SYSTEM.md §5.1:
   604→   604→//   • drag="x" with elastic constraints, card follows finger 1:1.
   605→   605→//   • swipe-left past 80 px (or 500 px/s velocity) → commit-delete:
   606→   606→//     card slides fully off-screen (200ms ease-in), then onDelete() fires
   607→   607→//     (which surfaces the Undo snackbar + collapses the row).
   608→   608→//   • swipe-right past 80 px (or 500 px/s velocity) → toggle read/unread,
   609→   609→//     card springs back to rest (no removal).
   610→   610→//   • background revealed underneath: red-500 with trash icon for delete,
   611→   611→//     emerald-500 with mail/mail-open icon swap for read/unread.
   612→   612→//   • icon scales up as |drag| grows.
   613→   613→//   • prefers-reduced-motion → drag disabled entirely; hover buttons still work.
   614→   614→//   • tap (no drag) → onOpen(). drag → suppress click.
   615→   615→function MessageListItem({
   616→   616→  msg, active, selected, fresh, selectMode, isSelected, onToggleSelect, onOpen, onToggleRead, onToggleStar, onDelete, onReport, onForward,
   617→   617→}: {
   618→   618→  msg: MessageSummary
   619→   619→  active: boolean
   620→   620→  selected: boolean
   621→   621→  fresh: boolean
   622→   622→  selectMode?: boolean
   623→   623→  isSelected?: boolean
   624→   624→  onToggleSelect?: () => void
   625→   625→  onOpen: () => void
   626→   626→  onToggleRead: () => void
   627→   627→  onToggleStar: () => void
   628→   628→  onDelete: () => void
   629→   629→  onReport: (reason: string, category: string) => void
   630→   630→  onForward?: () => void
   631→   631→}) {
   632→   632→  const cat = CATEGORY_META[msg.category] || CATEGORY_META.general
   633→   633→  const hasAuthIssue = msg.spf !== 'pass' || msg.dkim !== 'pass' || msg.dmarc !== 'pass'
   634→   634→
   635→   635→  const reduceMotion = useReducedMotion()
   636→   636→  const dragX = useMotionValue(0)
   637→   637→  const controls = useAnimationControls()
   638→   638→
   639→   639→  // Long-press context menu state
   640→   640→  const [showContextMenu, setShowContextMenu] = useState(false)
   641→   641→  const longPress = useLongPress({ onLongPress: () => setShowContextMenu(true) })
   642→   642→
   643→   643→  // Track whether a real drag happened so we can suppress the click that
   644→   644→  // follows. Framer Motion fires `onClick` even after a drag ends if the
   645→   645→  // pointer-up happens over the same element; we want a tap (no movement) to
   646→   646→  // open the message, and a drag to NOT open it.
   647→   647→  const draggedRef = useRef(false)
   648→   648→  const pointerDownRef = useRef<{ x: number; y: number } | null>(null)
   649→   649→
   650→   650→  // ---- drag x → background opacity + icon scale mappings ----
   651→   651→  // Delete background: 0 at rest → 1 when dragged to -120px.
   652→   652→  const deleteBgOpacity = useTransform(dragX, [-120, -40], [1, 0], { clamp: true })
   653→   653→  // Read/unread background: 0 at rest → 1 when dragged to +120px.
   654→   654→  const readBgOpacity = useTransform(dragX, [40, 120], [0, 1], { clamp: true })
   655→   655→  // Trash icon scales from 0.8 → 1.4 as the user swipes left past the threshold.
   656→   656→  const deleteIconScale = useTransform(dragX, [-120, -40], [1.4, 0.8], { clamp: true })
   657→   657→  // Mail icon scales from 0.8 → 1.4 as the user swipes right past the threshold.
   658→   658→  const readIconScale = useTransform(dragX, [40, 120], [0.8, 1.4], { clamp: true })
   659→   659→
   660→   660→  const COMMIT_THRESHOLD = 80 // px
   661→   661→  const VELOCITY_THRESHOLD = 500 // px/s
   662→   662→
   663→   663→  const handleDragEnd = async (_e: unknown, info: { offset: { x: number }; velocity: { x: number } }) => {
   664→   664→    // Clear the drag flag (the click that fires right after will check it).
   665→   665→    // Defer clearing by a tick so the click event sees `draggedRef = true`.
   666→   666→    setTimeout(() => { draggedRef.current = false }, 0)
   667→   667→
   668→   668→    const offset = info.offset.x
   669→   669→    const velocity = info.velocity.x
   670→   670→
   671→   671→    if (offset < -COMMIT_THRESHOLD || velocity < -VELOCITY_THRESHOLD) {
   672→   672→      // Swipe left → delete.
   673→   673→      // Slide the card fully off-screen (200ms ease-in per spec), then call
   674→   674→      // onDelete() which optimistically removes the message from the store
   675→   675→      // and triggers the row's exit-animation (height → 0).
   676→   676→      await controls.start({
   677→   677→        x: '-120%',
   678→   678→        opacity: 0,
   679→   679→        transition: { duration: 0.2, ease: 'easeIn' },
   680→   680→      })
   681→   681→      onDelete()
   682→   682→      // The parent <motion.li> unmounts (because removeMessage removes the
   683→   683→      // message from the store), so we don't need to reset dragX/controls here.
   684→   684→    } else if (offset > COMMIT_THRESHOLD || velocity > VELOCITY_THRESHOLD) {
   685→   685→      // Swipe right → toggle read/unread, then spring back to rest.
   686→   686→      onToggleRead()
   687→   687→      await controls.start({
   688→   688→        x: 0,
   689→   689→        transition: { type: 'spring', stiffness: 500, damping: 30 },
   690→   690→      })
   691→   691→    } else {
   692→   692→      // Spring back to rest (no commit).
   693→   693→      await controls.start({
   694→   694→        x: 0,
   695→   695→        transition: { type: 'spring', stiffness: 500, damping: 30 },
   696→   696→      })
   697→   697→    }
   698→   698→  }
   699→   699→
   700→   700→  const handleClick = () => {
   701→   701→    if (draggedRef.current) return
   702→   702→    // Suppress click after a long-press (the context menu opened)
   703→   703→    if (longPress.didLongPress.current) {
   704→   704→      return
   705→   705→    }
   706→   706→    onOpen()
   707→   707→  }
   708→   708→
   709→   709→  const handlePointerDown = (e: React.PointerEvent<HTMLButtonElement>) => {
   710→   710→    pointerDownRef.current = { x: e.clientX, y: e.clientY }
   711→   711→  }
   712→   712→  const handlePointerUp = (e: React.PointerEvent<HTMLButtonElement>) => {
   713→   713→    // If the pointer moved less than ~5 px since pointer-down, this was a tap,
   714→   714→    // not a drag. Framer Motion will fire onClick naturally; we just need to
   715→   715→    // make sure `draggedRef` reflects the truth.
   716→   716→    const down = pointerDownRef.current
   717→   717→    pointerDownRef.current = null
   718→   718→    if (down) {
   719→   719→      const dx = Math.abs(e.clientX - down.x)
   720→   720→      const dy = Math.abs(e.clientY - down.y)
   721→   721→      if (dx < 5 && dy < 5) {
   722→   722→        draggedRef.current = false
   723→   723→      }
   724→   724→    }
   725→   725→  }
   726→   726→
   727→   727→  return (
   728→   728→    <motion.li
   729→   729→      layout
   730→   730→      data-msg-id={msg.id}
   731→   731→      initial={fresh ? { opacity: 0, y: -16 } : false}
   732→   732→      animate={{ opacity: 1, y: 0 }}
   733→   733→      exit={{ opacity: 0, height: 0, marginBottom: 0, overflow: 'hidden' }}
   734→   734→      transition={{ type: 'spring', stiffness: 400, damping: 30 }}
   735→   735→      className="relative overflow-hidden"
   736→   736→    >
   737→   737→      {/* ---- Swipe action backgrounds (revealed underneath the card) ---- */}
   738→   738→      {/* Delete (left swipe) */}
   739→   739→      <div
   740→   740→        className="absolute inset-0 bg-red-500 flex items-center justify-start pl-6 pointer-events-none"
   741→   741→        aria-hidden="true"
   742→   742→      >
   743→   743→        <motion.div
   744→   744→          style={{ opacity: deleteBgOpacity, scale: deleteIconScale }}
   745→   745→          className="flex flex-col items-center gap-1 text-white"
   746→   746→        >
   747→   747→          <Trash2 className="h-5 w-5" />
   748→   748→          <span className="text-[10px] font-semibold uppercase tracking-wide">Delete</span>
   749→   749→        </motion.div>
   750→   750→      </div>
   751→   751→      {/* Read/unread (right swipe) */}
   752→   752→      <div
   753→   753→        className="absolute inset-0 bg-emerald-500 flex items-center justify-end pr-6 pointer-events-none"
   754→   754→        aria-hidden="true"
   755→   755→      >
   756→   756→        <motion.div
   757→   757→          style={{ opacity: readBgOpacity, scale: readIconScale }}
   758→   758→          className="flex flex-col items-center gap-1 text-white"
   759→   759→        >
   760→   760→          {msg.isRead ? <Mail className="h-5 w-5" /> : <MailOpen className="h-5 w-5" />}
   761→   761→          <span className="text-[10px] font-semibold uppercase tracking-wide">
   762→   762→            {msg.isRead ? 'Unread' : 'Read'}
   763→   763→          </span>
   764→   764→        </motion.div>
   765→   765→      </div>
   766→   766→
   767→   767→      {/* ---- Draggable card ---- */}
   768→   768→      <motion.div
   769→   769→        drag={reduceMotion ? false : 'x'}
   770→   770→        dragConstraints={{ left: 0, right: 0 }}
   771→   771→        dragElastic={0.5}
   772→   772→        dragMomentum={false}
   773→   773→        style={{ x: dragX }}
   774→   774→        animate={controls}
   775→   775→        onDragStart={() => { draggedRef.current = true }}
   776→   776→        onDragEnd={handleDragEnd}
   777→   777→        whileDrag={{ cursor: 'grabbing' }}
   778→   778→        className={cn(
   779→   779→          'relative group cursor-pointer bg-card transition-colors touch-pan-y',
   780→   780→          active
   781→   781→            ? 'bg-emerald-500/15 ring-1 ring-inset ring-emerald-500/50 shadow-sm'
   782→   782→            : selected
   783→   783→              ? 'bg-emerald-500/8 ring-1 ring-inset ring-emerald-500/30'
   784→   784→              : 'hover:bg-accent/40',
   785→   785→          isSelected && 'bg-emerald-500/10 ring-1 ring-inset ring-emerald-500/40',
   786→   786→          fresh && 'bg-emerald-500/5'
   787→   787→        )}
   788→   788→      >
   789→   789→        <button
   790→   790→          onClick={handleClick}
   791→   791→          onPointerDown={(e) => { handlePointerDown(e); longPress.bind.onPointerDown(e) }}
   792→   792→          onPointerMove={longPress.bind.onPointerMove}
   793→   793→          onPointerUp={(e) => { handlePointerUp(e); longPress.bind.onPointerUp(e) }}
   794→   794→          onPointerLeave={longPress.bind.onPointerLeave}
   795→   795→          onPointerCancel={longPress.bind.onPointerCancel}
   796→   796→          style={longPress.isLongPressing ? { transform: 'scale(0.98)', transition: 'transform 0.15s ease' } : undefined}
   797→   797→          className="w-full text-left p-3 flex gap-3 min-w-0"
   798→   798→          aria-label={`Open message: ${msg.subject} from ${msg.fromName}`}
   799→   799→          aria-haspopup={showContextMenu ? 'menu' : undefined}
   800→   800→          aria-expanded={showContextMenu}
   801→   801→        >
   802→   802→          {selectMode && (
   803→   803→            <div
   804→   804→              className={cn(
   805→   805→                'grid h-5 w-5 shrink-0 place-items-center rounded border-2 transition-colors mt-2.5',
   806→   806→                isSelected ? 'bg-emerald-500 border-emerald-500' : 'border-muted-foreground/30'
   807→   807→              )}
   808→   808→              onClick={(e) => { e.stopPropagation(); onToggleSelect?.() }}
   809→   809→            >
   810→   810→              {isSelected && <Check className="h-3 w-3 text-white" strokeWidth={3} />}

   972→   972→function UndoSnackbar({
   973→   973→  toastId,
   974→   974→  subject,
   975→   975→  onUndo,
   976→   976→}: {
   977→   977→  toastId: string | number
   978→   978→  subject: string
   979→   979→  onUndo: () => void
   980→   980→}) {
   981→   981→  const [progress, setProgress] = useState(100)
   982→   982→  useEffect(() => {
   983→   983→    const DURATION = 5000
   984→   984→    const start = performance.now()
   985→   985→    let raf = 0
   986→   986→    const tick = (now: number) => {
   987→   987→      const elapsed = now - start
   988→   988→      const remaining = Math.max(0, DURATION - elapsed)
   989→   989→      setProgress((remaining / DURATION) * 100)
   990→   990→      if (remaining > 0) raf = requestAnimationFrame(tick)
   991→   991→    }

   993→   993→    return () => cancelAnimationFrame(raf)
   994→   994→  }, [])
