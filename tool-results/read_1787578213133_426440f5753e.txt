   375→// ---------------- Customize Dialog ----------------
   376→function CustomizeDialog({
   377→  open, onOpenChange, domains, lifetimeOptions, categories, onCreated,
   378→}: {
   379→  open: boolean
   380→  onOpenChange: (v: boolean) => void
   381→  domains: Array<{ domain: string; label: string; badge: string; popular: boolean }>
   382→  lifetimeOptions: Array<{ value: number; label: string; default?: boolean }>
   383→  categories: Array<{ value: string; label: string; desc: string }>
   384→  onCreated: (inbox: Inbox) => void
   385→}) {
   386→  const [localPart, setLocalPart] = useState('')
   387→  const [domain, setDomain] = useState(domains[0]?.domain || 'studentbox.in')
   388→  const [lifetime, setLifetime] = useState(lifetimeOptions.find(o => o.default)?.value || 10)
   389→  const [category, setCategory] = useState('general')
   390→  const [burnOnRead, setBurnOnRead] = useState(false)
   391→  const [checkState, setCheckState] = useState<'idle' | 'checking' | 'available' | 'taken'>('idle')
   392→  const [checkReason, setCheckReason] = useState('')
   393→  const queryClient = useQueryClient()
   394→
   395→  useEffect(() => {
   396→    if (open) {
   397→      setLocalPart('')
   398→      setDomain(domains[0]?.domain || 'studentbox.in')
   399→      setLifetime(lifetimeOptions.find(o => o.default)?.value || 10)
   400→      setCategory('general')
   401→      setBurnOnRead(false)
   402→      setCheckState('idle')
   403→      setCheckReason('')
   404→    }
   405→  }, [open, domains, lifetimeOptions])
   406→
   407→  useEffect(() => {
   408→    if (!localPart || localPart.length < 3) {
   409→      setCheckState('idle')
   410→      setCheckReason('')
   411→      return
   412→    }
   413→    setCheckState('checking')
   414→    const handle = setTimeout(async () => {
   415→      try {
   416→        const res = await api.checkAlias(localPart, domain)
   417→        if (res.available) {
   418→          setCheckState('available')
   419→          setCheckReason('')
   420→        } else {
   421→          setCheckState('taken')
   422→          setCheckReason(res.reason || 'Not available')
   423→        }
   424→      } catch {
   425→        setCheckState('idle')
   426→      }
   427→    }, 500)
   428→    return () => clearTimeout(handle)
   429→  }, [localPart, domain])
   430→
   431→  const createMutation = useMutation({
   432→    mutationFn: () =>
   433→      api.createInbox({
   434→        domain,
   435→        lifetimeMinutes: lifetime,
   436→        category,
   437→        customLocalPart: localPart,
   438→        burnOnRead,
   439→      }),
   440→    onSuccess: (data) => {
   441→      queryClient.invalidateQueries({ queryKey: ['stats'] })
   442→      onCreated(data.inbox)
   443→    },
   444→    onError: (e: Error) => toast.error(e.message),
   445→  })
   446→
   447→  const canSubmit = checkState === 'available' && !createMutation.isPending
   448→
   449→  return (
   450→    <Dialog open={open} onOpenChange={onOpenChange}>
   451→      <DialogContent className="sm:max-w-lg">
   452→        <DialogHeader>
   453→          <DialogTitle className="flex items-center gap-2"><Sparkles className="h-4 w-4 text-emerald-500" /> Customize your address</DialogTitle>
