     3→/**
     4→ * App Lock — section + lock screen + auto-lock hook
     5→ *
     6→ * Local-only device-UI protection. The server has NO knowledge of the PIN.
     7→ *
     8→ * Setup: user picks a 4-6 digit PIN → we derive a PBKDF2-SHA256 key (100k
     9→ * iterations) with a random 16-byte salt, then AES-GCM-encrypt a known marker
    10→ * string. The encrypted marker (ciphertext + salt + iv) + pinLength +
    11→ * autoLockDelay + biometricEnabled live in localStorage under
    12→ * `studenttemp_applock_data`. The boolean enabled flag also lives under
    13→ * `studenttemp_applock` (mirrored by the Zustand store).
    14→ *
    15→ * Unlock: user re-enters PIN → re-derive key → try AES-GCM decrypt of the
    16→ * stored ciphertext → if plaintext === marker, success.
    17→ *
    18→ * Biometric: WebAuthn platform authenticator (Touch ID / Face ID / Windows
    19→ * Hello). We register a discoverable (resident) credential once at setup; on
    20→ * unlock we call navigator.credentials.get() with userVerification:'required'.
    21→ * The OS verifies the user; we trust the OS. Biometric is a UX shortcut — the
    22→ * PIN remains the cryptographic factor and is always available as fallback.
    23→ *
    24→ * Auto-lock: Page Visibility API. When the tab is hidden, we record a
    25→ * timestamp; on resume, if hidden-for > autoLockDelay, lock immediately (so the
    26→ * lock screen is present with zero flash-of-unlocked-content).
    27→ *
    28→ * Cool-down: after every 5 failed PIN attempts, lock the pad for 15s → 30s →
    29→ * 60s → 5min, escalating with each successive lockout. Counter persists in
    30→ * localStorage so closing the tab does NOT reset it.
    31→ *
    32→ * Forgot PIN: clears the local encrypted blob only. Server-side inboxes are
    33→ * unaffected.
    34→ *
    35→ * Spec: PRD SECURITY.md "App Lock Flow" + MOTION-SYSTEM.md §9.
    36→ */

   116→// =============================================================================
   117→// Crypto helpers (real Web Crypto API)

   120→function bufToB64(buf: ArrayBuffer | Uint8Array): string {
   121→  const bytes = buf instanceof Uint8Array ? buf : new Uint8Array(buf)
   122→  let s = ''
   123→  const chunk = 0x8000
   124→  for (let i = 0; i < bytes.length; i += chunk) {
   125→    s += String.fromCharCode.apply(null, Array.from(bytes.subarray(i, i + chunk)) as unknown as number[])
   126→  }
   127→  return btoa(s)
   128→}
   129→
   130→function b64ToBuf(s: string): Uint8Array<ArrayBuffer> {
   131→  const bin = atob(s)
   132→  const bytes = new Uint8Array(new ArrayBuffer(bin.length))
   133→  for (let i = 0; i < bin.length; i++) bytes[i] = bin.charCodeAt(i)
   134→  return bytes
   135→}
   136→
   137→function randomBytes(n: number): Uint8Array<ArrayBuffer> {
   138→  const arr = new Uint8Array(new ArrayBuffer(n))
   139→  crypto.getRandomValues(arr)
   140→  return arr
   141→}
   142→
   143→async function deriveKey(pin: string, salt: Uint8Array<ArrayBuffer>): Promise<CryptoKey> {
   144→  const enc = new TextEncoder()
   145→  const baseKey = await crypto.subtle.importKey(
   146→    'raw', enc.encode(pin), { name: 'PBKDF2' }, false, ['deriveKey']
   147→  )
   148→  return crypto.subtle.deriveKey(
   149→    { name: 'PBKDF2', salt, iterations: PBKDF2_ITERATIONS, hash: 'SHA-256' },
   150→    baseKey,
   151→    { name: 'AES-GCM', length: KEY_LENGTH_BITS },
   152→    false,
   153→    ['encrypt', 'decrypt']
   154→  )
   155→}
   156→
   157→async function encryptMarker(pin: string): Promise<{ salt: string; iv: string; ciphertext: string; pinLength: PinLength }> {
   158→  const salt = randomBytes(SALT_BYTES)
   159→  const iv = randomBytes(IV_BYTES)
   160→  const key = await deriveKey(pin, salt)
   161→  const enc = new TextEncoder()
   162→  const ct = await crypto.subtle.encrypt({ name: 'AES-GCM', iv }, key, enc.encode(UNLOCK_MARKER))
   163→  return {
   164→    salt: bufToB64(salt),
   165→    iv: bufToB64(iv),
   166→    ciphertext: bufToB64(ct),
   167→    pinLength: pin.length as PinLength,
   168→  }
   169→}
   170→
   171→async function verifyPin(pin: string, data: AppLockData): Promise<boolean> {
   172→  try {
   173→    const salt = b64ToBuf(data.salt)
   174→    const iv = b64ToBuf(data.iv)
   175→    const ct = b64ToBuf(data.ciphertext)
   176→    const key = await deriveKey(pin, salt)
   177→    const dec = await crypto.subtle.decrypt({ name: 'AES-GCM', iv }, key, ct)
   178→    const text = new TextDecoder().decode(dec)
   179→    return text === UNLOCK_MARKER
   180→  } catch {
   181→    return false
   182→  }
   183→}

   189→function loadData(): AppLockData | null {
   190→  if (typeof window === 'undefined') return null
   191→  try {
   192→    const raw = localStorage.getItem(LS_DATA_KEY)
   193→    if (!raw) return null
   194→    const parsed = JSON.parse(raw)
   195→    if (parsed?.version !== 1) return null
   196→    return parsed as AppLockData
   197→  } catch {
   198→    return null
   199→  }
   200→}
   201→
   202→function saveData(data: AppLockData | null): void {
   203→  if (typeof window === 'undefined') return
   204→  if (data === null) localStorage.removeItem(LS_DATA_KEY)
   205→  else localStorage.setItem(LS_DATA_KEY, JSON.stringify(data))
   206→}

   212→type BiometricStatus = 'available' | 'unavailable' | 'unsupported'

   214→async function getBiometricStatus(): Promise<BiometricStatus> {
   215→  if (typeof window === 'undefined' || !window.PublicKeyCredential) return 'unsupported'
   216→  try {
   217→    const fn = (PublicKeyCredential as unknown as {
   218→      isUserVerifyingPlatformAuthenticatorAvailable?: () => Promise<boolean>
   219→    }).isUserVerifyingPlatformAuthenticatorAvailable
   220→    if (!fn) return 'unavailable'
   221→    const ok = await fn.call(PublicKeyCredential)
   222→    return ok ? 'available' : 'unavailable'
   223→  } catch {
   224→    return 'unavailable'
   225→  }
   226→}

   228→/** Registers a discoverable (resident) platform credential to use as a local
   229→ *  unlock token. The OS verifies the user; we trust the OS. */
   230→async function registerBiometric(): Promise<string | null> {
   231→  try {
   232→    const challenge = randomBytes(32)
   233→    const userId = randomBytes(16)
   234→    const cred = (await navigator.credentials.create({
   235→      publicKey: {
   236→        challenge,
   237→        rp: { name: 'StudentTemp App Lock' },
   238→        user: {
   239→          id: userId,
   240→          name: 'app-lock-user',
   241→          displayName: 'App Lock',
   242→        },
   243→        pubKeyCredParams: [
   244→          { type: 'public-key', alg: -7 },    // ES256
   245→          { type: 'public-key', alg: -257 },  // RS256
   246→        ],
   247→        authenticatorSelection: {
   248→          authenticatorAttachment: 'platform',
   249→          userVerification: 'required',
   250→          residentKey: 'required',
   251→          requireResidentKey: true,
   252→        },
   253→        timeout: 60_000,
   254→        attestation: 'none',
   255→      },
   256→    })) as PublicKeyCredential | null
   257→    if (!cred) return null
   258→    return bufToB64(cred.rawId)
   259→  } catch {
   260→    return null
   261→  }
   262→}
   263→
   264→async function unlockWithBiometric(credentialId: string): Promise<boolean> {
   265→  try {
   266→    const challenge = randomBytes(32)
   267→    const assertion = await navigator.credentials.get({
   268→      publicKey: {
   269→        challenge,
   270→        timeout: 60_000,
   271→        userVerification: 'required',
   272→        allowCredentials: [{
   273→          type: 'public-key',
   274→          id: b64ToBuf(credentialId),
   275→          transports: ['internal'],
   276→        }],
   277→      },
   278→    })
   279→    return !!assertion
   280→  } catch {
   281→    return false
   282→  }
   283→}

   417→// =============================================================================
   418→// CooldownTimer — countdown shown during progressive lockout
   419→// =============================================================================
   420→
   421→function CooldownTimer({ until, onExpire }: { until: number; onExpire: () => void }) {
   422→  const [remaining, setRemaining] = useState(() => Math.max(0, Math.ceil((until - Date.now()) / 1000)))
   423→
   424→  useEffect(() => {
   425→    const tick = () => {
   426→      const r = Math.max(0, Math.ceil((until - Date.now()) / 1000))
   427→      setRemaining(r)
   428→      if (r <= 0) { onExpire() }
   429→    }
   430→    tick()
   431→    const id = setInterval(tick, 500)
   432→    return () => clearInterval(id)
   433→  }, [until, onExpire])
   434→
   435→  const mins = Math.floor(remaining / 60)
   436→  const secs = remaining % 60
   437→  const display = mins > 0 ? `${mins}:${String(secs).padStart(2, '0')}` : `${secs}s`
   438→
   439→  return (
   440→    <motion.div
   441→      initial={{ opacity: 0, y: 4 }}
   442→      animate={{ opacity: 1, y: 0 }}
   443→      className="flex flex-col items-center gap-1 rounded-xl border border-amber-500/30 bg-amber-500/10 px-4 py-2.5 text-center"
   444→      role="status"
   445→      aria-live="polite"
   446→    >
   447→      <div className="flex items-center gap-1.5 text-amber-700 dark:text-amber-400">
   448→        <Clock className="h-4 w-4" />
   449→        <span className="text-sm font-medium">Too many attempts</span>
   450→      </div>
   451→      <div className="text-2xl font-bold tabular-nums text-amber-700 dark:text-amber-400">
   452→        {display}
   453→      </div>
   454→      <div className="text-xs text-muted-foreground">Try again in a moment</div>
   455→    </motion.div>
   456→  )
   457→}

   463→interface LockScreenProps {
   464→  /** Optional callback fired when the unlock transition finishes. */
   465→  onUnlocked?: () => void
   466→}
   467→
   468→export function LockScreen({ onUnlocked }: LockScreenProps) {
   469→  const isLocked = useAppStore((s) => s.isLocked)
   470→  const setLocked = useAppStore((s) => s.setLocked)
   471→  const appLockEnabled = useAppStore((s) => s.appLockEnabled)
   472→  const sound = useSound()
   473→
   474→  const [data, setData] = useState<AppLockData | null>(null)
   475→  const [mounted, setMounted] = useState(false)
   476→  const [entered, setEntered] = useState('')
   477→  const [shake, setShake] = useState(false)
   478→  const [verifying, setVerifying] = useState(false)
   479→  const [unlocked, setUnlocked] = useState(false)
   480→  const [bioBusy, setBioBusy] = useState(false)
   481→
   482→  // Avoid SSR/CSR mismatch: only act on the client.
   483→  useEffect(() => { setMounted(true) }, [])
   484→
   485→  // Hydrate lock data from localStorage.
   486→  useEffect(() => {
   487→    if (!mounted) return
   488→    setData(loadData())
   489→  }, [mounted, isLocked])
   490→
   491→  // Reset PIN entry on mount.
   492→  useEffect(() => {
   493→    if (isLocked) { setEntered(''); setShake(false); setUnlocked(false) }
   494→  }, [isLocked])
   495→
   496→  const pinLength: PinLength = data?.pinLength ?? 4
   497→
   498→  const inCooldown = !!data?.cooldownUntil && data.cooldownUntil > Date.now()
   499→  const cooldownMs = data?.cooldownUntil ?? 0
   500→
   501→  const handleUnlocked = useCallback(() => {
   502→    setUnlocked(true)
   503→    sound.playUnlock()
   504→    // Persist: clear failed attempts + cooldown.
   505→    if (data) {
   506→      const next: AppLockData = { ...data, failedAttempts: 0, cooldownUntil: null }
   507→      saveData(next); setData(next)
   508→    }
   509→    // Let the success animation play, then drop the lock for real.
   510→    const t = setTimeout(() => {
   511→      setLocked(false)
   512→      onUnlocked?.()
   513→    }, 320)
   514→    return () => clearTimeout(t)
   515→  }, [data, onUnlocked, setLocked, sound])
   516→
   517→  const submit = useCallback(async (pin: string) => {
   518→    if (!data || verifying || inCooldown) return
   519→    setVerifying(true)
   520→    const ok = await verifyPin(pin, data)
   521→    setVerifying(false)
   522→    if (ok) {
   523→      handleUnlocked()
   524→    } else {
   525→      // Shake, then clear + increment attempts.
   526→      setShake(true)
   527→      sound.playError()
   528→      setTimeout(() => setShake(false), 320)
   529→      setEntered('')
   530→      const attempts = data.failedAttempts + 1
   531→      const next: AppLockData = { ...data, failedAttempts: attempts }
   532→      // Every MAX_FAILED_BEFORE_COOLDOWN failures → escalate.
   533→      if (attempts % MAX_FAILED_BEFORE_COOLDOWN === 0) {
   534→        const idx = Math.min(
   535→          Math.floor(attempts / MAX_FAILED_BEFORE_COOLDOWN) - 1,
   536→          COOLDOWN_LADDER.length - 1
   537→        )
   538→        next.cooldownUntil = Date.now() + COOLDOWN_LADDER[idx] * 1000
   539→        toast.error(`Too many attempts. Locked for ${formatDuration(COOLDOWN_LADDER[idx])}.`)
   540→      } else {
   541→        toast.error('Wrong PIN', {
   542→          description: `${MAX_FAILED_BEFORE_COOLDOWN - (attempts % MAX_FAILED_BEFORE_COOLDOWN)} attempts before cooldown`,
   543→        })
   544→      }
   545→      saveData(next); setData(next)
   546→    }
   547→  }, [data, verifying, inCooldown, sound, handleUnlocked])
   548→
   549→  const tryBiometric = useCallback(async () => {
   550→    if (!data?.biometricCredentialId || bioBusy) return
   551→    setBioBusy(true)
   552→    const ok = await unlockWithBiometric(data.biometricCredentialId)
   553→    setBioBusy(false)
   554→    if (ok) {
   555→      handleUnlocked()
   556→    } else {
   557→      toast('Biometric unavailable', {
   558→        description: 'Use your PIN to unlock',
   559→      })
   560→    }
   561→  }, [data, bioBusy, handleUnlocked])
   562→
   563→  // Auto-trigger biometric when the lock screen first appears (one-shot).
   564→  const autoBioTriedRef = useRef(false)
   565→  useEffect(() => {
   566→    if (!mounted || !isLocked || !data) return
   567→    if (autoBioTriedRef.current) return
   568→    autoBioTriedRef.current = true
   569→    if (data.biometricEnabled && data.biometricCredentialId) {
   570→      // Defer so the lock screen paints first (zero-flash).
   571→      const t = setTimeout(() => { void tryBiometric() }, 300)
   572→      return () => clearTimeout(t)
   573→    }
   574→  }, [mounted, isLocked, data, tryBiometric])
   575→
   576→  // Reset the auto-bio ref when the lock re-engages.
   577→  useEffect(() => {
   578→    if (isLocked) autoBioTriedRef.current = false
   579→  }, [isLocked])
   580→
   581→  // ---- focus trap (keyboard nav within the lock screen) ----
   582→  const containerRef = useRef<HTMLDivElement | null>(null)
   583→  useEffect(() => {
   584→    if (!isLocked || !mounted) return
   585→    const node = containerRef.current
   586→    if (!node) return
   587→    const focusables = () => Array.from(
   588→      node.querySelectorAll<HTMLElement>('button:not([disabled]), [tabindex]:not([tabindex="-1"])')
   589→    )
   590→    // Focus the first digit by default.
   591→    const t = setTimeout(() => {
   592→      const fs = focusables()
   593→      // Prefer the biometric button (if present) so user can just press Enter;
   594→      // otherwise focus the digit "5" in the middle.
   595→      const target = fs.find((el) => el.getAttribute('aria-label') === 'Digit 5')
   596→        ?? fs.find((el) => el.getAttribute('aria-label') === 'Unlock with biometric')
   597→        ?? fs[0]
   598→      target?.focus()
   599→    }, 60)
   600→    const handler = (e: globalThis.KeyboardEvent) => {
   601→      if (e.key !== 'Tab') return
   602→      const fs = focusables()
   603→      if (fs.length === 0) return
   604→      const first = fs[0]
   605→      const last = fs[fs.length - 1]
   606→      const active = document.activeElement as HTMLElement | null
   607→      if (e.shiftKey && active === first) { e.preventDefault(); last.focus() }
   608→      else if (!e.shiftKey && active === last) { e.preventDefault(); first.focus() }
   609→    }
   610→    node.addEventListener('keydown', handler)
   611→    return () => { clearTimeout(t); node.removeEventListener('keydown', handler) }
   612→  }, [isLocked, mounted])
   613→
   614→  // Physical keyboard: digits, Backspace, Enter.
   615→  useEffect(() => {
   616→    if (!isLocked || !mounted) return
   617→    const handler = (e: globalThis.KeyboardEvent) => {
   618→      if (inCooldown || verifying || unlocked) return
   619→      if (/^[0-9]$/.test(e.key)) {
   620→        e.preventDefault()
   621→        setEntered((cur) => cur.length < pinLength ? cur + e.key : cur)
   622→      } else if (e.key === 'Backspace') {
   623→        e.preventDefault()
   624→        setEntered((cur) => cur.slice(0, -1))
   625→      } else if (e.key === 'Enter') {
   626→        e.preventDefault()
   627→        setEntered((cur) => {
   628→          if (cur.length === pinLength) void submit(cur)
   629→          return cur
   630→        })
   631→      }
   632→    }
   633→    window.addEventListener('keydown', handler)
   634→    return () => window.removeEventListener('keydown', handler)
   635→  }, [isLocked, mounted, inCooldown, verifying, unlocked, pinLength, submit])
   636→
   637→  // Auto-submit when the user fills the expected PIN length.
   638→  useEffect(() => {
   639→    if (entered.length === pinLength && pinLength > 0) {
   640→      void submit(entered)
   641→    }
   642→  }, [entered, pinLength, submit])
   643→
   644→  // ---- render ----
   645→  if (!mounted) return null
   646→
   647→  // Don't show the lock screen if App Lock isn't enabled at all.
   648→  const shouldShow = isLocked && appLockEnabled
   649→  if (!shouldShow) return null
   650→
   651→  // If data isn't loaded yet, show a blank backdrop (zero-flash protection).
   652→  const ready = !!data
   653→
   654→  return (
   655→    <AnimatePresence>
   656→      {shouldShow && (
   657→        <motion.div
   658→          ref={containerRef}
   659→          role="dialog"
   660→          aria-modal="true"
   661→          aria-label="App locked. Enter your PIN to continue."
   662→          className="fixed inset-0 z-[60] flex flex-col items-center justify-center bg-background/95 backdrop-blur-md px-4"
   663→          initial={{ opacity: 0 }}
   664→          animate={unlocked ? { opacity: 0, scale: 1.03 } : { opacity: 1, scale: 1 }}
   665→          exit={{ opacity: 0, scale: 1.03 }}
   666→          transition={{ duration: 0.32, ease: [0.16, 1, 0.3, 1] }}
   667→        >
   668→          {/* Brand lock icon */}
   669→          <motion.div
   670→            initial={{ scale: 0.8, opacity: 0 }}
   671→            animate={{ scale: 1, opacity: 1 }}
   672→            transition={{ delay: 0.05, type: 'spring', stiffness: 320, damping: 26 }}
   673→            className="mb-5 grid h-16 w-16 place-items-center rounded-2xl bg-gradient-to-br from-emerald-500 to-cyan-500 text-white shadow-lg shadow-emerald-500/30"
   674→          >
   675→            <Lock className="h-7 w-7" />
   676→          </motion.div>
   677→
   678→          <h2 className="text-lg font-semibold tracking-tight">StudentTemp is locked</h2>
   679→          <p className="mt-1 text-sm text-muted-foreground">
   680→            {ready ? 'Enter your PIN to continue' : 'Preparing secure unlock…'}
   681→          </p>
   682→
   683→          {/* Dot indicators */}
   684→          <div className="my-6 h-6">
   685→            {ready && (
   686→              <PinDots count={pinLength} filled={entered.length} shake={shake} error={shake} />
   687→            )}
   688→          </div>
   689→
   690→          {/* Cooldown OR pin pad */}
   691→          <div className="w-full max-w-xs">
   692→            {inCooldown ? (
   693→              <CooldownTimer until={cooldownMs} onExpire={() => {
   694→                if (!data) return
   695→                const next = { ...data, cooldownUntil: null }
   696→                saveData(next); setData(next)
   697→              }} />
   698→            ) : ready ? (
   699→              <PinPad
   700→                onDigit={(d) => setEntered((cur) => cur.length < pinLength ? cur + d : cur)}
   701→                onDelete={() => setEntered((cur) => cur.slice(0, -1))}
   702→                onSubmit={() => { if (entered.length === pinLength) void submit(entered) }}
   703→                onBiometric={tryBiometric}
   704→                showBiometric={!!data.biometricEnabled && !!data.biometricCredentialId}
   705→                biometricBusy={bioBusy}
   706→                disabled={verifying}

   740→    </AnimatePresence>
   741→  )

   750→function SetupDialog({
   751→  open, onOpenChange, bioStatus, onConfirm,
   752→}: {
   753→  open: boolean
   754→  onOpenChange: (v: boolean) => void
   755→  bioStatus: BiometricStatus
   756→  onConfirm: (pin: string, useBiometric: boolean) => Promise<void>
   757→}) {
   758→  const [step, setStep] = useState<SetupStep>('enter')
   759→  const [firstPin, setFirstPin] = useState('')
   760→  const [confirmPin, setConfirmPin] = useState('')
   761→  const [useBiometric, setUseBiometric] = useState(true)
   762→  const [busy, setBusy] = useState(false)
   763→  const [shake, setShake] = useState(false)
   764→  const sound = useSound()
   765→
   766→  // Reset on close.
   767→  useEffect(() => {
   768→    if (!open) {
   769→      const t = setTimeout(() => {
   770→        setStep('enter'); setFirstPin(''); setConfirmPin(''); setUseBiometric(true); setBusy(false); setShake(false)
   771→      }, 250)
   772→      return () => clearTimeout(t)
   773→    }
   774→  }, [open])
   775→
   776→  const dotCount = useMemo(() => {
   777→    // Show min 4 dots; grow up to 6 as the user types more digits.
   778→    return Math.max(PIN_MIN, Math.min(firstPin.length, PIN_MAX))
   779→  }, [firstPin])
   780→
   781→  const handleEnterDigit = (d: string) => {
   782→    if (step === 'enter') {
   783→      setFirstPin((cur) => cur.length < PIN_MAX ? cur + d : cur)
   784→    } else if (step === 'confirm') {
   785→      setConfirmPin((cur) => cur.length < firstPin.length ? cur + d : cur)
   786→    }
   787→  }
   788→  const handleDelete = () => {
   789→    if (step === 'enter') setFirstPin((cur) => cur.slice(0, -1))
   790→    else setConfirmPin((cur) => cur.slice(0, -1))
   791→  }
   792→  const handleSubmit = () => {
   793→    if (step === 'enter') {
   794→      if (firstPin.length < PIN_MIN) {
   795→        toast.error('PIN must be at least 4 digits')
   796→        return
   797→      }
   798→      setStep('confirm')
   799→    } else if (step === 'confirm') {
   800→      if (confirmPin !== firstPin) {
   801→        setShake(true); sound.playError()
   802→        setTimeout(() => { setShake(false); setConfirmPin('') }, 320)
   803→        toast.error('PINs do not match', { description: 'Try entering it again' })
   804→        return
   805→      }
   806→      // Skip biometric step if unavailable.
   807→      if (bioStatus === 'available') setStep('biometric')
   808→      else void finalize(false)
   809→    } else if (step === 'biometric') {
   810→      void finalize(useBiometric)
   811→    }
   812→  }
   813→
   814→  const finalize = async (withBio: boolean) => {
   815→    setBusy(true)
   816→    await onConfirm(firstPin, withBio)
   817→    setBusy(false)
   818→    onOpenChange(false)
   819→  }
   820→
   821→  const currentPin = step === 'enter' ? firstPin : confirmPin
   822→  const currentLen = step === 'enter' ? firstPin.length : confirmPin.length
   823→  const expectedLen = step === 'enter' ? Math.max(PIN_MIN, firstPin.length) : firstPin.length
   824→  const title =
   825→    step === 'enter' ? 'Set your PIN' :
   826→    step === 'confirm' ? 'Confirm your PIN' :
   827→    'Enable biometric?'
   828→
   829→  return (
   830→    <Dialog open={open} onOpenChange={onOpenChange}>
   831→      <DialogContent className="max-w-sm">
   832→        <DialogHeader>
   833→          <DialogTitle className="flex items-center gap-2">
   834→            <KeyRound className="h-4 w-4 text-primary" />
   835→            {title}
   836→          </DialogTitle>
   837→          <DialogDescription>
   838→            {step === 'enter' && 'Pick a 4-6 digit PIN. This PIN is never sent to the server.'}
   839→            {step === 'confirm' && 'Re-enter the PIN to make sure it\'s correct.'}
   840→            {step === 'biometric' && 'Use Touch ID / Face ID / fingerprint to unlock faster? You can always fall back to your PIN.'}
   841→          </DialogDescription>
   842→        </DialogHeader>
   843→
   844→        {step !== 'biometric' ? (
   845→          <>
   846→            <div className="flex justify-center py-2">
   847→              <PinDots count={step === 'enter' ? dotCount : expectedLen} filled={currentLen} shake={shake} error={shake} />
   848→            </div>
   849→            <PinPad
   850→              onDigit={handleEnterDigit}
   851→              onDelete={handleDelete}
   852→              onSubmit={handleSubmit}
   853→              actionVariant={currentLen >= PIN_MIN ? 'submit' : 'delete'}
   854→              disabled={busy}
   855→            />
   856→            {step === 'enter' && (
   857→              <p className="text-center text-[11px] text-muted-foreground">
   858→                {firstPin.length < PIN_MIN
   859→                  ? `${PIN_MIN - firstPin.length} more digit${PIN_MIN - firstPin.length === 1 ? '' : 's'} minimum`
   860→                  : `${firstPin.length}-digit PIN`}
   861→              </p>
   862→            )}
   863→          </>
   864→        ) : (
   865→          <div className="space-y-4">
   866→            <div className="flex items-start gap-3 rounded-lg border border-border/60 bg-muted/30 p-3">
   867→              <Fingerprint className="h-5 w-5 text-primary shrink-0 mt-0.5" />
   868→              <div className="text-sm">
   869→                <div className="font-medium">Biometric unlock</div>
   870→                <div className="text-xs text-muted-foreground mt-0.5">
   871→                  Your device will prompt you to authenticate. The PIN you just set always works as a fallback.
   872→                </div>
   873→              </div>
   874→            </div>
   875→            <div className="flex gap-2">
   876→              <Button variant="outline" className="flex-1" onClick={() => void finalize(false)} disabled={busy}>
   877→                Skip
   878→              </Button>
   879→              <Button className="flex-1 gap-2" onClick={() => void finalize(true)} disabled={busy}>
   880→                {busy ? <Loader2 className="h-4 w-4 animate-spin" /> : <Fingerprint className="h-4 w-4" />}
   881→                Enable biometric
   882→              </Button>
   883→            </div>
   884→          </div>
   885→        )}
   886→
   887→        {/* Step indicator */}
   888→        <div className="flex items-center justify-center gap-1.5 pt-1">
   889→          {(['enter', 'confirm', 'biometric'] as SetupStep[])
   890→            .filter((s) => bioStatus === 'available' || s !== 'biometric')
   891→            .map((s) => (
   892→              <span
   893→                key={s}
   894→                className={cn(
   895→                  'h-1.5 w-6 rounded-full transition-colors',
   896→                  s === step ? 'bg-primary' : (stepOrder(step) > stepOrder(s) ? 'bg-primary/40' : 'bg-muted')
   897→                )}
   898→              />
   899→            ))}
   900→        </div>
   901→      </DialogContent>
   902→    </Dialog>
   903→  )
   904→}

   906→function stepOrder(s: SetupStep): number {
   907→  return s === 'enter' ? 0 : s === 'confirm' ? 1 : 2

   914→function ChangePINDialog({
   915→  open, onOpenChange, currentData, onChanged,
   916→}: {
   917→  open: boolean
   918→  onOpenChange: (v: boolean) => void
   919→  currentData: AppLockData
   920→  onChanged: (next: AppLockData) => void
   921→}) {
   922→  const [phase, setPhase] = useState<'old' | 'new' | 'confirm'>('old')
   923→  const [oldPin, setOldPin] = useState('')
   924→  const [newPin, setNewPin] = useState('')
   925→  const [confirmPin, setConfirmPin] = useState('')
   926→  const [busy, setBusy] = useState(false)
   927→  const [shake, setShake] = useState(false)
   928→  const sound = useSound()
   929→
   930→  useEffect(() => {
   931→    if (!open) {
   932→      const t = setTimeout(() => {
   933→        setPhase('old'); setOldPin(''); setNewPin(''); setConfirmPin(''); setBusy(false); setShake(false)
   934→      }, 250)
   935→      return () => clearTimeout(t)
   936→    }
   937→  }, [open])
   938→
   939→  const expectedLen =
   940→    phase === 'old' ? currentData.pinLength :
   941→    phase === 'new' ? Math.max(PIN_MIN, newPin.length) :
   942→    newPin.length
   943→
   944→  const currentPin = phase === 'old' ? oldPin : phase === 'new' ? newPin : confirmPin
   945→  const currentLen = currentPin.length
   946→  const maxLen = phase === 'old' ? currentData.pinLength : PIN_MAX
   947→
   948→  const handleDigit = (d: string) => {
   949→    if (phase === 'old') setOldPin((c) => c.length < currentData.pinLength ? c + d : c)
   950→    else if (phase === 'new') setNewPin((c) => c.length < PIN_MAX ? c + d : c)
   951→    else setConfirmPin((c) => c.length < newPin.length ? c + d : c)
   952→  }
   953→  const handleDelete = () => {
   954→    if (phase === 'old') setOldPin((c) => c.slice(0, -1))
   955→    else if (phase === 'new') setNewPin((c) => c.slice(0, -1))
   956→    else setConfirmPin((c) => c.slice(0, -1))
   957→  }
   958→  const handleSubmit = async () => {
   959→    if (phase === 'old') {
   960→      if (oldPin.length !== currentData.pinLength) return
   961→      setBusy(true)
   962→      const ok = await verifyPin(oldPin, currentData)
   963→      setBusy(false)
   964→      if (!ok) {
   965→        setShake(true); sound.playError()
   966→        setTimeout(() => { setShake(false); setOldPin('') }, 320)
   967→        toast.error('Current PIN is incorrect')
   968→        return
   969→      }
   970→      setPhase('new')
   971→    } else if (phase === 'new') {
   972→      if (newPin.length < PIN_MIN) {
   973→        toast.error('PIN must be at least 4 digits')
   974→        return
   975→      }
   976→      setPhase('confirm')
   977→    } else if (phase === 'confirm') {
   978→      if (confirmPin !== newPin) {
   979→        setShake(true); sound.playError()
   980→        setTimeout(() => { setShake(false); setConfirmPin('') }, 320)
   981→        toast.error('PINs do not match')
   982→        return
   983→      }
   984→      setBusy(true)
   985→      const enc = await encryptMarker(newPin)
   986→      const next: AppLockData = {
   987→        ...currentData,
   988→        ...enc,
   989→        failedAttempts: 0,
   990→        cooldownUntil: null,
   991→        createdAt: new Date().toISOString(),
   992→      }
   993→      saveData(next)
   994→      setBusy(false)
   995→      onChanged(next)
   996→      toast.success('PIN changed')
   997→      onOpenChange(false)
   998→    }
   999→  }
  1000→
  1001→  const title =
  1002→    phase === 'old' ? 'Enter current PIN' :
  1003→    phase === 'new' ? 'Set a new PIN' :
  1004→    'Confirm new PIN'
  1005→
  1006→  return (
  1007→    <Dialog open={open} onOpenChange={onOpenChange}>
  1008→      <DialogContent className="max-w-sm">
  1009→        <DialogHeader>
  1010→          <DialogTitle className="flex items-center gap-2">
  1011→            <RefreshCw className="h-4 w-4 text-primary" />
  1012→            {title}
  1013→          </DialogTitle>
  1014→          <DialogDescription>
  1015→            {phase === 'old' && 'Verify it\'s you first.'}
  1016→            {phase === 'new' && 'Pick a new 4-6 digit PIN.'}
  1017→            {phase === 'confirm' && 'Re-enter the new PIN to confirm.'}
  1018→          </DialogDescription>
  1019→        </DialogHeader>
  1020→        <div className="flex justify-center py-2">
  1021→          <PinDots count={expectedLen} filled={currentLen} shake={shake} error={shake} />
  1022→        </div>
  1023→        <PinPad
  1024→          onDigit={handleDigit}
  1025→          onDelete={handleDelete}
  1026→          onSubmit={handleSubmit}
  1027→          actionVariant={currentLen >= (phase === 'old' ? currentData.pinLength : PIN_MIN) ? 'submit' : 'delete'}
  1028→          disabled={busy}
  1029→        />
  1030→        {busy && (
  1031→          <div className="flex items-center justify-center text-xs text-muted-foreground gap-2">
  1032→            <Loader2 className="h-3.5 w-3.5 animate-spin" /> Verifying…
  1033→          </div>
  1034→        )}
  1035→      </DialogContent>
  1036→    </Dialog>

  1040→// =============================================================================
  1041→// useAutoLock — Page Visibility API hook (exported for AppShell to use)
  1042→// =============================================================================
  1043→
  1044→/**
  1045→ * Watches the document visibility state and locks the app when the tab is
  1046→ * backgrounded for longer than the configured delay. Also exposes a manual
  1047→ * `lockNow` callback. No-op when App Lock is disabled.
  1048→ *
  1049→ * Mount this once at the app root (AppShell) so the lock triggers regardless
  1050→ * of which section is currently active.
  1051→ */
  1052→export function useAutoLock() {
  1053→  const appLockEnabled = useAppStore((s) => s.appLockEnabled)
  1054→  const isLocked = useAppStore((s) => s.isLocked)
  1055→  const setLocked = useAppStore((s) => s.setLocked)
  1056→  const hiddenAtRef = useRef<number | null>(null)
  1057→
  1058→  useEffect(() => {
  1059→    if (!appLockEnabled) return
  1060→    if (typeof document === 'undefined') return
  1061→
  1062→    const onVisChange = () => {
  1063→      if (document.hidden) {
  1064→        hiddenAtRef.current = Date.now()
  1065→      } else {
  1066→        const hiddenAt = hiddenAtRef.current
  1067→        hiddenAtRef.current = null
  1068→        if (hiddenAt == null) return
  1069→        const data = loadData()
  1070→        const delay = data?.autoLockDelay ?? 120
  1071→        if (delay <= 0) return // Never
  1072→        const elapsed = Date.now() - hiddenAt
  1073→        if (elapsed >= delay * 1000) {
  1074→          // Lock while still hidden→visible transition; the lock screen mounts
  1075→          // before the user can see the unlocked content (zero-flash).
  1076→          setLocked(true)
  1077→        }
  1078→      }
  1079→    }
  1080→
  1081→    const onBlur = () => {
  1082→      // Some browsers fire `blur` without `visibilitychange` on quick tab switches.
  1083→      // We rely on visibilitychange as the source of truth; blur is a fallback.
  1084→      if (document.visibilityState === 'hidden') return
  1085→      // No-op: page is still visible, just lost focus.
  1086→    }
  1087→
  1088→    document.addEventListener('visibilitychange', onVisChange)
  1089→    window.addEventListener('blur', onBlur)
  1090→    return () => {
  1091→      document.removeEventListener('visibilitychange', onVisChange)
  1092→      window.removeEventListener('blur', onBlur)
  1093→    }
  1094→  }, [appLockEnabled, setLocked])
  1095→

  1601→function formatDuration(seconds: number): string {
  1602→  if (seconds <= 0) return 'Never'
  1603→  if (seconds < 60) return `${seconds}s`
  1604→  if (seconds < 3600) {
  1605→    const m = Math.floor(seconds / 60)
  1606→    return m === 1 ? '1 minute' : `${m} minutes`
  1607→  }
  1608→  const h = Math.floor(seconds / 3600)
  1609→  return h === 1 ? '1 hour' : `${h} hours`
  1610→}
