   481→   481→function GlobalSearchInline() {

   483→   483→  const setOpen = useAppStore((s) => s.setGlobalSearchOpen)

   490→   490→    queryKey: ['global-search', query],

   492→   492→      if (!query || query.length < 2) return { results: [] as Array<Record<string, unknown>>, total: 0 }
   493→   493→      const res = await fetch(`/api/search?q=${encodeURIComponent(query)}`)

   505→   505→    setOpen(false)

   507→   507→
   508→   508→  const results = data?.results || []

   587→   587→// ---------- Search result highlighting ----------
   588→   588→function highlightMatch(text: string, query: string): ReactNode {
   589→   589→  if (!query || !text) return text
   590→   590→  const lower = text.toLowerCase()
   591→   591→  const q = query.toLowerCase()
   592→   592→  const idx = lower.indexOf(q)
   593→   593→  if (idx === -1) return text
   594→   594→  return (
   595→   595→    <>
   596→   596→      {text.slice(0, idx)}
   597→   597→      <mark className="bg-emerald-500/25 text-emerald-800 dark:text-emerald-200 rounded px-0.5 font-semibold">
   598→   598→        {text.slice(idx, idx + q.length)}
   599→   599→      </mark>
   600→   600→      {text.slice(idx + q.length)}
   601→   601→    </>
   602→   602→  )
   603→   603→}
