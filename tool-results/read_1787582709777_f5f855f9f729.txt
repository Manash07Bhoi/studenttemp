    38→function sanitizeHtml(html: string): string {
    39→  if (!html) return '<p style="color:#999;font-family:Arial">(empty body)</p>'
    40→  let out = html
    41→  out = out.replace(/<script[\s\S]*?<\/script>/gi, '')
    42→  out = out.replace(/<style[\s\S]*?<\/style>/gi, '')
    43→  out = out.replace(/\son\w+="[^"]*"/gi, '')
    44→  out = out.replace(/\son\w+='[^']*'/gi, '')
    45→  out = out.replace(/javascript:/gi, '')
    46→  out = out.replace(/(<img[^>]+)src=["']https?:\/\/[^"']*["']/gi, '$1data-src-blocked="true"')
    47→  out = out.replace(/<link[^>]+href=["']https?:\/\/[^"']*["'][^>]*>/gi, '')
    48→  out = out.replace(/<iframe[\s\S]*?<\/iframe>/gi, '')
    49→  return `<base target="_blank"><style>body{font-family:Arial,Helvetica,sans-serif;color:#1a1a1a;background:#fff;line-height:1.6;margin:0;padding:12px}a{color:#0ea5e9}img{max-width:100%;height:auto}</style>${out}`
    50→}

    52→function formatTime(iso: string): string {
    53→  const d = new Date(iso)
    54→  const now = new Date()
    55→  const diff = now.getTime() - d.getTime()
    56→  if (diff < 60_000) return 'just now'
    57→  if (diff < 3_600_000) return `${Math.floor(diff / 60_000)}m ago`
    58→  if (d.toDateString() === now.toDateString()) return d.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' })
    59→  return d.toLocaleDateString([], { month: 'short', day: 'numeric' })
    60→}

   406→// ---------------- Message Reader ----------------
   407→function MessageReader({
   408→  messageSummary, onBack, onDelete, onToggleStar, onReport,
   409→}: {
   410→  messageSummary: MessageSummary
   411→  onBack: () => void
   412→  onDelete: () => void
   413→  onToggleStar: () => void
   414→  onReport: (reason: string, category: string) => void
   415→}) {
   416→  const [full, setFull] = useState<MessageFull | null>(null)
   417→  const [loading, setLoading] = useState(true)
   418→  const [showRaw, setShowRaw] = useState(false)
   419→  const [showAuth, setShowAuth] = useState(false)
   420→  const [showReport, setShowReport] = useState(false)
   421→  const [imagesLoaded, setImagesLoaded] = useState(false)
   422→
   423→  useEffect(() => {
   424→    let active = true
   425→    setLoading(true)
   426→    setFull(null)
   427→    setImagesLoaded(false)
   428→    api.getMessage(messageSummary.id).then((data) => {
   429→      if (active) {
   430→        setFull(data.message)
   431→        setLoading(false)
   432→      }
   433→    }).catch(() => setLoading(false))
   434→    return () => { active = false }
   435→  }, [messageSummary.id])
   436→
   437→  const cat = CATEGORY_META[messageSummary.category] || CATEGORY_META.general
   438→  const authFail = messageSummary.spf !== 'pass' || messageSummary.dkim !== 'pass' || messageSummary.dmarc !== 'pass'
   439→  const spoofed = messageSummary.fromName !== messageSummary.fromEmail &&
   440→    /paypal|bank|secure|verify|official|government/i.test(messageSummary.fromName)
   441→
   442→  const sanitizedHtml = useMemo(() => {
   443→    if (!full?.bodyHtml) return ''
   444→    let out = full.bodyHtml
   445→    if (imagesLoaded) {
   446→      // restore blocked image srcs
   447→      out = out.replace(/(<img[^>]+)data-src-blocked=["']true["']/gi, '$1')
   448→    } else {
   449→      out = out.replace(/(<img[^>]+)src=["']https?:\/\/[^"']*["']/gi, '$1data-src-blocked="true"')
   450→    }
   451→    return sanitizeHtml(out)
   452→  }, [full?.bodyHtml, imagesLoaded])
   453→
   454→  return (
   455→    <motion.div
   456→      initial={{ opacity: 0, x: 8 }} animate={{ opacity: 1, x: 0 }} exit={{ opacity: 0, x: -8 }}
   457→      transition={{ duration: 0.2 }}
   458→      className="flex flex-col h-full min-h-[55vh] lg:min-h-[65vh]"
   459→    >
   460→      {/* Reader header */}
   461→      <div className="border-b border-border/60 p-4 space-y-3">
   462→        <div className="flex items-center gap-2">
   463→          <Button variant="ghost" size="sm" onClick={onBack} className="gap-1 lg:hidden">
   464→            <ArrowLeft className="h-4 w-4" /> Back
   465→          </Button>
   466→          <div className="flex-1" />
   467→          <Button variant="ghost" size="icon" className="h-8 w-8" onClick={onToggleStar} title={messageSummary.isStarred ? 'Unstar' : 'Star'}>
   468→            <Star className={cn('h-4 w-4', messageSummary.isStarred && 'text-amber-500 fill-amber-500')} />
   469→          </Button>
   470→          <Button variant="ghost" size="icon" className="h-8 w-8 text-red-500 hover:bg-red-500/10" onClick={onDelete} title="Delete">
   471→            <Trash2 className="h-4 w-4" />
   472→          </Button>
   473→          <DropdownMenu>
   474→            <DropdownMenuTrigger asChild>
   475→              <Button variant="ghost" size="icon" className="h-8 w-8" title="More">
   476→                <ChevronRight className="h-4 w-4 rotate-90" />
   477→              </Button>
   478→            </DropdownMenuTrigger>
   479→            <DropdownMenuContent align="end">
   480→              <DropdownMenuItem onClick={() => setShowAuth(!showAuth)} className="gap-2">
   481→                <ShieldCheck className="h-3.5 w-3.5" /> {showAuth ? 'Hide' : 'Show'} security panel
   482→              </DropdownMenuItem>
   483→              <DropdownMenuItem onClick={() => setShowRaw(!showRaw)} className="gap-2">
   484→                <Mail className="h-3.5 w-3.5" /> {showRaw ? 'HTML view' : 'Plain text view'}
   485→              </DropdownMenuItem>
   486→              <DropdownMenuSeparator />
   487→              <DropdownMenuItem onClick={() => setShowReport(true)} className="gap-2 text-red-600">
   488→                <Flag className="h-3.5 w-3.5" /> Report this message
   489→              </DropdownMenuItem>
   490→            </DropdownMenuContent>
   491→          </DropdownMenu>
   492→        </div>
   493→
   494→        <h2 className="text-lg font-bold leading-tight">{messageSummary.subject}</h2>
   495→
   496→        <div className="flex items-start gap-3">
   497→          <div className="grid h-10 w-10 shrink-0 place-items-center rounded-full bg-gradient-to-br from-emerald-400 to-cyan-500 text-white font-bold text-sm">
   498→            {messageSummary.fromName[0]?.toUpperCase() || '?'}
   499→          </div>
   500→          <div className="flex-1 min-w-0">
   501→            <div className="flex items-center gap-2 flex-wrap">
   502→              <span className="font-semibold text-sm">{messageSummary.fromName}</span>
   503→              {spoofed && (
   504→                <Badge variant="outline" className="bg-amber-500/10 text-amber-600 border-amber-500/20 text-[10px] gap-1">
   505→                  <AlertTriangle className="h-2.5 w-2.5" /> Display name looks suspicious
   506→                </Badge>
   507→              )}
   508→              <Badge variant="outline" className={cn('text-[10px]', cat.color)}>{cat.label}</Badge>
   509→            </div>
   510→            <p className="mt-0.5 text-xs text-muted-foreground font-mono break-all">&lt;{messageSummary.fromEmail}&gt;</p>
   511→            <p className="mt-0.5 text-xs text-muted-foreground flex items-center gap-1">
   512→              <Clock className="h-3 w-3" /> {new Date(messageSummary.receivedAt).toLocaleString()}
   513→            </p>
   514→          </div>
   515→        </div>
   516→
   517→        {/* Auth panel */}
   518→        <AnimatePresence>
   519→          {showAuth && (
   520→            <motion.div
   521→              initial={{ opacity: 0, height: 0 }} animate={{ opacity: 1, height: 'auto' }} exit={{ opacity: 0, height: 0 }}
   522→              className="overflow-hidden"
   523→            >
   524→              <div className="grid grid-cols-3 gap-2 pt-2">
   525→                <AuthChip label="SPF" value={messageSummary.spf} />
   526→                <AuthChip label="DKIM" value={messageSummary.dkim} />
   527→                <AuthChip label="DMARC" value={messageSummary.dmarc} />
   528→              </div>
   529→              {authFail && (
   530→                <div className="mt-2 rounded-lg bg-amber-500/10 border border-amber-500/20 p-2.5 text-xs text-amber-700 dark:text-amber-300 flex gap-2">
   531→                  <ShieldAlert className="h-4 w-4 shrink-0 mt-0.5" />
   532→                  <span>One or more authentication checks failed. Be cautious — the sender's identity could not be fully verified.</span>
   533→                </div>
   534→              )}
   535→            </motion.div>
   536→          )}
   537→        </AnimatePresence>
   538→
   539→        {/* Attachments */}
   540→        {full && full.attachments && full.attachments.length > 0 && (
   541→          <div className="flex flex-wrap gap-2 pt-1">
   542→            {full.attachments.map((a, i) => (
   543→              <div key={i} className="flex items-center gap-2 rounded-lg border border-border/60 bg-muted/40 px-2.5 py-1.5 text-xs">
   544→                <Paperclip className="h-3.5 w-3.5 text-muted-foreground" />
   545→                <span className="font-medium">{a.name}</span>
   546→                <span className="text-muted-foreground">{(a.size / 1024).toFixed(0)} KB</span>
   547→              </div>
   548→            ))}
   549→          </div>
   550→        )}
   551→      </div>
   552→
   553→      {/* External resources blocked banner */}
   554→      {messageSummary.externalResourcesBlocked > 0 && !imagesLoaded && (
   555→        <motion.div
   556→          initial={{ y: -20, opacity: 0 }} animate={{ y: 0, opacity: 1 }}
   557→          className="border-b border-amber-500/20 bg-amber-500/5 px-4 py-2.5 flex items-center gap-2 text-xs text-amber-700 dark:text-amber-300"
   558→        >
   559→          <Ban className="h-4 w-4 shrink-0" />
   560→          <span className="flex-1">{messageSummary.externalResourcesBlocked} external resource{messageSummary.externalResourcesBlocked !== 1 ? 's' : ''} blocked for your safety.</span>
   561→          <button
   562→            onClick={() => setImagesLoaded(true)}
   563→            className="font-medium underline-offset-2 hover:underline"
   564→          >
   565→            Load anyway
   566→          </button>
   567→        </motion.div>
   568→      )}
   569→
   570→      {/* Body */}
   571→      <ScrollArea className="flex-1 scrollbar-thin">
   572→        <div className="p-4">
   573→          {loading ? (
   574→            <div className="space-y-3">
   575→              <Skeleton className="h-4 w-3/4" />
   576→              <Skeleton className="h-4 w-full" />
   577→              <Skeleton className="h-4 w-5/6" />
   578→              <Skeleton className="h-24 w-full" />
   579→              <Skeleton className="h-4 w-2/3" />
   580→            </div>
   581→          ) : showRaw ? (
   582→            <pre className="whitespace-pre-wrap break-words text-sm font-mono text-foreground/80">
   583→              {full?.bodyText}
   584→            </pre>
   585→          ) : (
   586→            <motion.div
   587→              initial={{ opacity: 0, y: 8 }} animate={{ opacity: 1, y: 0 }}
   588→              transition={{ duration: 0.25 }}
   589→            >
   590→              <iframe
   591→                title="message body"
   592→                srcDoc={sanitizedHtml}
   593→                className="w-full min-h-[300px] border-0"
   594→                sandbox="allow-same-origin"
   595→              />
   596→            </motion.div>
   597→          )}
   598→        </div>
   599→      </ScrollArea>
   600→
   601→      {/* Report dialog */}
   602→      <Dialog open={showReport} onOpenChange={setShowReport}>
   603→        <DialogContent className="sm:max-w-md">
   604→          <DialogHeader>
   605→            <DialogTitle className="flex items-center gap-2"><Flag className="h-4 w-4 text-red-500" /> Report this message</DialogTitle>
   606→            <DialogDescription>
   607→              Flag this message for review. The sender is not notified.
   608→            </DialogDescription>
   609→          </DialogHeader>
   610→          <ReportForm onSubmit={(reason, category) => { onReport(reason, category); setShowReport(false) }} />
   611→        </DialogContent>
   612→      </Dialog>
   613→    </motion.div>
   614→  )
   615→}

   617→function ReportForm({ onSubmit }: { onSubmit: (reason: string, category: string) => void }) {
   618→  const [category, setCategory] = useState('spam')
   619→  const [reason, setReason] = useState('')
   620→  return (
   621→    <div className="space-y-4 py-2">
   622→      <div className="grid grid-cols-2 gap-2">
   623→        {[
   624→          { v: 'spam', l: 'Spam / Promo' },
   625→          { v: 'phishing', l: 'Phishing / Fraud' },
   626→          { v: 'abuse', l: 'Abuse / Harassment' },
   627→          { v: 'other', l: 'Other' },
   628→        ].map((o) => (
   629→          <button
   630→            key={o.v}
   631→            onClick={() => setCategory(o.v)}
   632→            className={cn(
   633→              'rounded-lg border px-3 py-2 text-sm font-medium transition-colors text-left',
   634→              category === o.v ? 'border-primary bg-primary/10 text-primary' : 'border-border hover:bg-accent'
   635→            )}
   636→          >
   637→            {o.l}
   638→          </button>
   639→        ))}
   640→      </div>
   641→      <textarea
   642→        value={reason}
   643→        onChange={(e) => setReason(e.target.value)}
   644→        placeholder="Add details (optional)..."
   645→        className="w-full min-h-[80px] rounded-lg border border-border bg-background px-3 py-2 text-sm resize-none focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring"
   646→      />
   647→      <DialogFooter>
   648→        <Button onClick={() => onSubmit(reason, category)} className="gap-2">
   649→          <Flag className="h-4 w-4" /> Submit report

   651→      </DialogFooter>
   652→    </div>

   656→function AuthChip({ label, value }: { label: string; value: string }) {
   657→  const pass = value === 'pass'
   658→  const warn = value === 'softfail' || value === 'none'
   659→  return (
   660→    <div className={cn(
   661→      'rounded-lg border p-2 text-center',
   662→      pass && 'bg-emerald-500/10 border-emerald-500/20',
   663→      warn && 'bg-amber-500/10 border-amber-500/20',
   664→      !pass && !warn && 'bg-red-500/10 border-red-500/20',
   665→    )}>
   666→      <div className="text-[10px] uppercase tracking-wider text-muted-foreground">{label}</div>
   667→      <div className={cn(
   668→        'text-sm font-bold flex items-center justify-center gap-1',
   669→        pass && 'text-emerald-600 dark:text-emerald-400',
   670→        warn && 'text-amber-600 dark:text-amber-400',
   671→        !pass && !warn && 'text-red-600 dark:text-red-400',
   672→      )}>
   673→        {pass ? <ShieldCheck className="h-3 w-3" /> : <ShieldAlert className="h-3 w-3" />}
   674→        {value}
   675→      </div>
   676→    </div>
   677→  )
   678→}
